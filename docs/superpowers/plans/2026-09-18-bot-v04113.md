## Global Constraints

- 🔴 **Ни одного файла вне `bot/`.** Правка в `api-go/` означает, что задача уехала не в тот релиз.
- 🔴 **Никогда `tgbotapi.NewInlineKeyboardMarkup()` без строк** — сериализуется в `{"inline_keyboard":null}`, настоящий Telegram отвечает `Bad Request: field "inline_keyboard" must be of type Array`, кнопка становится мёртвой. Пустая клавиатура — только `keyboards.None()`. Это стережёт `TestNoEmptyKeyboardConstructor`.
- 🔴 **Весь текст — через `i18n.T(lang, ru, en)`.** Голый литерал в месте отправки ловит `internal/handlers/i18n_scan_test.go`.
- 🔴 **`callback_data` ≤ 64 байт.** UUID календаря — 36 символов, значит префикс не длиннее 27 байт.
- **Язык:** идентификаторы, комментарии и коммиты — English. Тексты для пользователя — обе строки в `i18n.T`.
- **Коммит** после каждой задачи, без со-автора Claude/Anthropic, `git add` конкретных файлов, никогда `-A`.
- **Проверка перед «готово»:** `cd bot && go build ./... && go vet ./... && go test ./...`
- **Порядок полей карточки фиксирован:** дата · время · название · повтор · календарь · напоминание · теги · цвет · заметка.

// fieldLine prints one characteristic of a card, ALWAYS.
//
// 🔴 Denis, 18.09: an empty field is printed as «нет», not skipped. A field
// that only appears when filled is invisible to the person who has never
// filled it — and that is how someone learns the app has no repeats, no tags
// and no calendars. Absence of a line is not absence of a feature.
func fieldLine(icon, label, value string) string {
	return fmt.Sprintf("%s %s %s\n", icon, label, value)
}

// orNone renders the empty value as a word, in the user's language.
func orNone(lang i18n.Lang, value string) string {
	if strings.TrimSpace(value) == "" {
		return i18n.T(lang, "нет", "none")
	}
	return value
}
```

package api

// CalendarDetail is a calendar plus the two things a screen needs and the plain
// Calendar does not carry: what the caller may do with it, and how many people
// are in it.
type CalendarDetail struct {
	ID         string `json:"id"`
	Name       string `json:"name"`
	Colour     string `json:"color"`
	Role       string `json:"role"`
	IsPersonal bool   `json:"is_personal"`
	Members    int    `json:"member_count"`
}

// Member is one person on a calendar.
type Member struct {
	UserID string `json:"user_id"`
	Email  string `json:"email"`
	Name   string `json:"name"`
	Role   string `json:"role"`
}

func (c *Client) CalendarsFull(token string) ([]CalendarDetail, error) {
	var resp struct {
		Data []CalendarDetail `json:"data"`
	}
	if err := c.get("/api/calendars", token, nil, &resp); err != nil {
		return nil, err
	}
	return resp.Data, nil
}

func (c *Client) CreateCalendar(token, name, colour string) (*Calendar, error) {
	body := map[string]any{"name": name}
	if colour != "" {
		body["color"] = colour
	}
	var resp struct {
		Data Calendar `json:"data"`
	}
	if err := c.post("/api/calendars", token, body, &resp); err != nil {
		return nil, err
	}
	return &resp.Data, nil
}

// UpdateCalendar sends ONLY the fields that changed.
//
// 🔴 The API's updateRequest carries *string for both (calendars/handlers.go:20),
// so an absent key means «leave it». Sending an empty `name` alongside a colour
// change would blank the title — pointers here are the difference.
func (c *Client) UpdateCalendar(token, id string, name, colour *string) error {
	body := map[string]any{}
	if name != nil {
		body["name"] = *name
	}
	if colour != nil {
		body["color"] = *colour
	}
	return c.patch("/api/calendars/"+id, token, body, nil)
}

func (c *Client) DeleteCalendar(token, id string) error {
	return c.del("/api/calendars/"+id, token)
}

func (c *Client) CalendarMembers(token, id string) ([]Member, error) {
	var resp struct {
		Data []Member `json:"data"`
	}
	if err := c.get("/api/calendars/"+id+"/members", token, nil, &resp); err != nil {
		return nil, err
	}
	return resp.Data, nil
}

func (c *Client) InviteByEmail(token, id, email, role string) error {
	return c.post("/api/calendars/"+id+"/invites", token,
		map[string]any{"email": email, "role": role}, nil)
}

func (c *Client) CalendarInviteLink(token, id, role string) (string, error) {
	var resp struct {
		Data struct {
			Token string `json:"token"`
		} `json:"data"`
	}
	if err := c.post("/api/calendars/"+id+"/invite-links", token,
		map[string]any{"role": role}, &resp); err != nil {
		return "", err
	}
	return resp.Data.Token, nil
}

func (c *Client) AcceptInviteLink(token, linkToken string) (*Calendar, error) {
	var resp struct {
		Data Calendar `json:"data"`
	}
	if err := c.post("/api/calendars/invite-links/accept", token,
		map[string]any{"token": linkToken}, &resp); err != nil {
		return nil, err
	}
	return &resp.Data, nil
}

// LeaveCalendar removes a member. Passing the caller's own id is how leaving
// works: RemoveMember (api-go calendars/members.go:284) starts with
// `leaving := actorID == targetID` and only demands ownership for anyone else.
func (c *Client) LeaveCalendar(token, id, userID string) error {
	return c.del("/api/calendars/"+id+"/members/"+userID, token)
}
```

// showCalendars lists every calendar the user can see.
func (h *Handler) showCalendars(chat int64, messageID int) {
	lang := h.langOf(chat)
	token, ok := h.tokenOf(chat)
	if !ok {
		return
	}
	cals, err := h.api.CalendarsFull(token)
	if err != nil {
		h.send(chat, i18n.T(lang, "Не смог получить календари, попробуй ещё раз.",

	}

	var b strings.Builder
	b.WriteString(i18n.T(lang, "<b>📁 Календари</b>\n\n", "<b>📁 Calendars</b>\n\n"))
	names := make([]string, 0, len(cals))
	ids := make([]string, 0, len(cals))
	for _, c := range cals {
		line := format.Escape(c.Name)
		if c.Members > 1 {
			line += fmt.Sprintf(" · %d", c.Members)
		}

		}
		b.WriteString("📁 " + line + "\n")
		names = append(names, c.Name)
		ids = append(ids, c.ID)
	}
	h.editOrSend(chat, messageID, b.String(), keyboards.CalendarList(lang, names, ids))
}

```go
// snoozeMinutes maps an action code to the delay it means.
//
// 🔴 The minutes live in the code, not in the payload. callback_data is capped
// at 64 bytes and a reminder id already spends 36 of them; a payload that grows
// with every new snooze option is a payload that will one day be truncated —
// and a truncated button is a message Telegram refuses to send at all.
func snoozeMinutes(code byte) int {
	switch code {
	case codeSnooze60:
		return 60
	default:
		return 10
	}
}
```

Ветка обработки `codeSnooze` в том же файле должна принимать оба новых кода и передавать `snoozeMinutes(code)` в `NotificationAction`. Старый `codeSnooze` **оставить принимаемым**: под уже отправленными напоминаниями на телефонах живут кнопки со старым кодом, и они обязаны продолжать работать.
